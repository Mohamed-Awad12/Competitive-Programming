void solve() {

    int n; cin >> n;
    vector<int> arr(n), pref(n);

    for (int i = 0; i < n; ++i) {
        cin >> arr[i];
        pref[i] = arr[i];
        if(i) pref[i] += pref[i - 1];
    }

    int dp[n][n];
    int opt[n][n];
    for (int i = 0; i < n; ++i) {
        dp[i][i] = 0;
        opt[i][i] = i;
    }

    for (int len = 2; len <= n; ++len) {
        for (int l = 0; l <= n - len; ++l) {
            int r = l + len - 1;
            int sum = pref[r];
            if(l) sum -= pref[l - 1];

            dp[l][r] = 1e18;
            for (int k = max(l, opt[l][r - 1]); k <= min(opt[l + 1][r], r - 1); ++k) {
                int curr = dp[l][k] + dp[k + 1][r] + sum;
                if(curr < dp[l][r]) {
                    dp[l][r] = curr;
                    opt[l][r] = k;
                }
            }
        }
    }

    cout << dp[0][n - 1] << '\n';
}
