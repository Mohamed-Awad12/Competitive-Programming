#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
using i128 = __int128_t;

// --------------------------------------------------
// Basic sums
// --------------------------------------------------

// 1 + 2 + ... + n
i128 sum1ToN(int64 n) {
    return (i128)n * (n + 1) / 2;
}

// 1² + 2² + ... + n²
i128 sumSquares(int64 n) {
    return (i128)n * (n + 1) * (2 * n + 1) / 6;
}

// 1³ + 2³ + ... + n³
i128 sumCubes(int64 n) {
    i128 sum = sum1ToN(n);
    return sum * sum;
}

// Sum from l to r.
i128 rangeSum(int64 l, int64 r) {
    if (l > r) return 0;
    return sum1ToN(r) - sum1ToN(l - 1);
}

// Number of integers from l to r.
int64 rangeCount(int64 l, int64 r) {
    if (l > r) return 0;
    return r - l + 1;
}

// --------------------------------------------------
// Arithmetic progression
// --------------------------------------------------

// a, a+d, a+2d, ...
i128 arithmeticNthTerm(int64 a, int64 d, int64 n) {
    return (i128)a + (i128)(n - 1) * d;
}

// Sum of first n terms.
i128 arithmeticSum(int64 a, int64 d, int64 n) {
    return (i128)n * (2 * (i128)a + (i128)(n - 1) * d) / 2;
}

// Sum when first and last terms are known.
i128 arithmeticSumFirstLast(int64 first, int64 last, int64 n) {
    return (i128)n * (first + (i128)last) / 2;
}

// --------------------------------------------------
// Geometric progression
// --------------------------------------------------

// a, ar, ar², ...
i128 geometricNthTerm(int64 a, int64 ratio, int64 n) {
    i128 result = a;

    for (int64 i = 1; i < n; i++) {
        result *= ratio;
    }

    return result;
}

// Sum: a + ar + ... + ar^(n-1)
// Use only when the result fits in __int128.
i128 geometricSum(int64 a, int64 ratio, int64 n) {
    if (n <= 0) return 0;
    if (ratio == 1) return (i128)a * n;

    i128 power = 1;
    i128 base = ratio;
    int64 exponent = n;

    while (exponent > 0) {
        if (exponent & 1) power *= base;
        base *= base;
        exponent >>= 1;
    }

    return (i128)a * (power - 1) / (ratio - 1);
}

// --------------------------------------------------
// Counting multiples
// --------------------------------------------------

// Number of multiples of x in [1, n].
int64 countMultiples(int64 n, int64 x) {
    if (n <= 0 || x <= 0) return 0;
    return n / x;
}

// Number of multiples of x in [l, r].
int64 countMultiplesInRange(int64 l, int64 r, int64 x) {
    if (l > r || x <= 0) return 0;

    return r / x - (l - 1) / x;
}

// Sum of multiples of x in [1, n].
i128 sumMultiples(int64 n, int64 x) {
    if (n <= 0 || x <= 0) return 0;

    int64 count = n / x;
    return (i128)x * count * (count + 1) / 2;
}

// Sum of multiples of x in [l, r].
i128 sumMultiplesInRange(int64 l, int64 r, int64 x) {
    if (l > r || x <= 0) return 0;

    return sumMultiples(r, x) - sumMultiples(l - 1, x);
}

// --------------------------------------------------
// Even and odd numbers
// --------------------------------------------------

// 2 + 4 + ... + 2n
i128 sumFirstNEven(int64 n) {
    return (i128)n * (n + 1);
}

// 1 + 3 + ... + (2n - 1)
i128 sumFirstNOdd(int64 n) {
    return (i128)n * n;
}

// Number of even values in [l, r].
int64 countEven(int64 l, int64 r) {
    if (l > r) return 0;
    return r / 2 - (l - 1) / 2;
}

// Number of odd values in [l, r].
int64 countOdd(int64 l, int64 r) {
    return rangeCount(l, r) - countEven(l, r);
}

// --------------------------------------------------
// Pairs and subarrays
// --------------------------------------------------

// Number of unordered pairs chosen from n elements.
i128 numberOfPairs(int64 n) {
    return (i128)n * (n - 1) / 2;
}

// Number of subarrays in an array of size n.
i128 numberOfSubarrays(int64 n) {
    return (i128)n * (n + 1) / 2;
}

// Number of subarrays of exact length k.
int64 subarraysOfLengthK(int64 n, int64 k) {
    if (k <= 0 || k > n) return 0;
    return n - k + 1;
}

// Number of non-empty subsequences.
// Warning: grows very quickly.
i128 numberOfNonEmptySubsequences(int64 n) {
    i128 result = 1;

    for (int64 i = 0; i < n; i++) {
        result *= 2;
    }

    return result - 1;
}

// --------------------------------------------------
// Grid formulas
// --------------------------------------------------

// Number of rectangles in an n × m grid of cells.
i128 numberOfRectangles(int64 n, int64 m) {
    return sum1ToN(n) * sum1ToN(m);
}

// Number of squares in an n × n grid.
i128 numberOfSquares(int64 n) {
    return sumSquares(n);
}

// Manhattan distance.
int64 manhattanDistance(
    int64 x1,
    int64 y1,
    int64 x2,
    int64 y2
) {
    return llabs(x1 - x2) + llabs(y1 - y2);
}

// --------------------------------------------------
// Divisibility helpers
// --------------------------------------------------

int64 floorDivision(int64 a, int64 b) {
    assert(b != 0);

    int64 quotient = a / b;
    int64 remainder = a % b;

    if (remainder != 0 && ((remainder > 0) != (b > 0))) {
        quotient--;
    }

    return quotient;
}

int64 ceilDivision(int64 a, int64 b) {
    assert(b != 0);
    return -floorDivision(-a, b);
}

// Ceiling of a / b when a >= 0 and b > 0.
int64 ceilDivPositive(int64 a, int64 b) {
    return a / b + (a % b != 0);
}

// Round x up to the nearest multiple of k.
int64 roundUpToMultiple(int64 x, int64 k) {
    return ceilDivision(x, k) * k;
}

// Round x down to the nearest multiple of k.
int64 roundDownToMultiple(int64 x, int64 k) {
    return floorDivision(x, k) * k;
}

// --------------------------------------------------
// GCD and LCM
// --------------------------------------------------

int64 gcdValue(int64 a, int64 b) {
    return std::gcd(a, b);
}

int64 lcmValue(int64 a, int64 b) {
    if (a == 0 || b == 0) return 0;

    return a / std::gcd(a, b) * b;
}

// --------------------------------------------------
// Printing __int128
// --------------------------------------------------

void printInt128(i128 value) {
    if (value == 0) {
        cout << 0;
        return;
    }

    if (value < 0) {
        cout << '-';
        value = -value;
    }

    string result;

    while (value > 0) {
        result.push_back(char('0' + value % 10));
        value /= 10;
    }

    reverse(result.begin(), result.end());
    cout << result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int64 n = 10;

    printInt128(sum1ToN(n));       // 55
    cout << '\n';

    printInt128(sumSquares(n));    // 385
    cout << '\n';

    printInt128(sumCubes(n));      // 3025
    cout << '\n';

    return 0;
}
