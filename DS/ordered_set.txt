#include <bits/stdc++.h>
#include <ext/pb_ds/assoc_container.hpp>
#include <ext/pb_ds/tree_policy.hpp>

using namespace std;
using namespace __gnu_pbds;

template<typename T>
using ordered_set = tree<
    T,
    null_type,
    less<T>,
    rb_tree_tag,
    tree_order_statistics_node_update
>;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    ordered_set<int> s;

    s.insert(10);
    s.insert(20);
    s.insert(30);
    s.insert(40);

    // Number of elements strictly smaller than x.
    cout << s.order_of_key(30) << '\n'; // 2

    // Iterator to the element at zero-based index k.
    cout << *s.find_by_order(2) << '\n'; // 30

    // Check whether an element exists.
    cout << s.count(20) << '\n'; // 1

    // Delete an element.
    s.erase(20);

    for (int value : s) {
        cout << value << ' ';
    }
    cout << '\n';

    return 0;
}
