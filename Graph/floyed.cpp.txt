  int n, m;
    cin >> n >> m;
    int dist[n + 5][n + 5];
    vector<array<int,3>>edges;

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (i == j) dist[i][j] = 0;
            else dist[i][j] = 1e15;
        }
    }


    for (int i = 0; i < m; i++) {
        int u, v,w;
        cin >> u >> v >> w;
        edges.push_back({w,u,v});
        dist[u][v] = min(dist[u][v], 1ll * w);
        dist[v][u] = min(dist[v][u], 1ll * w);
    }


    for (int k = 1; k <= n; k++) {
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]);
            }
        }
    }
